# Container settings that scripts/check_container_hygiene.py would refuse,
# kept on purpose.
#
# The check fails the Code Quality job for a runtime socket mount, an image
# that names no version, a privileged setting or a port published on every
# interface, in any tracked Compose file. List a finding here only when it is
# deliberate, as "rule  file  subject  # reason"; the subject is what the
# finding prints in brackets. An entry without a reason, or one that matches
# no finding any more, fails the check too.
#
# Of the Dockerfile rules only "download" can be listed here (a download that
# is verified some other way). An unverified pip or npm install, a base image
# without a digest and a download piped to a shell have no exception.

# The gateway is the platform's entry point: browsers, API clients and
# sensors on other machines reach it (docs/guides/ports.md).
port  docker-compose.yml  gateway:443   # HTTPS entry point of the platform
port  docker-compose.yml  gateway:80    # answers /health and redirects to HTTPS
port  docker-compose.yml  gateway:8080  # answers /health and redirects to HTTPS

# Reverse proxies of the standalone per-service stacks. Each is that stack's
# entry point; the service behind it is published on 127.0.0.1 only.
port  open-security-tools/docker-compose.yml         nginx:80   # reverse proxy, started only with --profile production
port  open-security-tools/docker-compose.yml         nginx:443  # reverse proxy, started only with --profile production
